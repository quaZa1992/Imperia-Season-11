
imp_state_destruction_effect = 
{
	if = {
		limit = { civilian_shelter > 0 }
		set_temp_variable = { shelter_levels_effect = building_level@civilian_shelter }
		multiply_temp_variable = { shelter_levels_effect = -0.08 }
		set_temp_variable = { shelter_sufficiency = 1 }
		add_to_temp_variable = { shelter_sufficiency = shelter_levels_effect }
		multiply_temp_variable = { casualty_count = shelter_sufficiency }
	}
	multiply_temp_variable = { casualty_count = 10 } #Optimization, gets rid of 980 loop iterations, maximum casualty = 40 Million People per state; If by one reason you need bigger limit then you did something wrong.
	for_loop_effect = {
		end = 5
		add_manpower = casualty_count
	}
	multiply_variable = { productivity_state_var = productivity_remain }
	clamp_variable = {
		var = productivity_state_var
		min = 200
	}
	
	## BUILDING EFFECTS
	
	# Civs
	
	set_temp_variable = { nuked_building_t = building_level@industrial_complex }
	multiply_temp_variable = { nuked_building_t = building_damages_n }
	round_temp_variable = nuked_building_t
	
	remove_building = { type = industrial_complex level = nuked_building_t }
	
	
	# Mils
	
	set_temp_variable = { nuked_building_t = building_level@arms_factory }

	# Office

	# Dockyard
