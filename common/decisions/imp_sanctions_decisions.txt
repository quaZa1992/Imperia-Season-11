imp_sanctions_decisions = 
{
	d_imp_check_sanctions_on = 
	{
		icon = GFX_decision_imp_list

		visible = 
		{
			is_ai = no
		}

		available = 
		{
			OR = { 
				NOT = {  check_variable = { imp_fin_sanc_by_us^0 = 0 } }
				NOT = {  check_variable = { imp_eco_sanc_by_us^0 = 0 } }
				NOT = {  check_variable = { imp_dip_sanc_by_us^0 = 0 } }
				NOT = {  check_variable = { imp_mil_sanc_by_us^0 = 0 } }
			}
		}

		complete_effect = 
		{
				country_event = Imp_sanctions.1
		}
	}

	d_imp_check_sanctions_by = 
	{
		icon = GFX_decision_imp_list

		visible = 
		{
			is_ai = no
		}

		available = 
		{
			OR = { 
				NOT = {  check_variable = { imp_fin_sanc_on_us^0 = 0 } }
				NOT = {  check_variable = { imp_eco_sanc_on_us^0 = 0 } }
				NOT = {  check_variable = { imp_dip_sanc_on_us^0 = 0 } }
				NOT = {  check_variable = { imp_mil_sanc_on_us^0 = 0 } }
			}
		}

		complete_effect = 
		{
				country_event = Imp_sanctions.2
		}
	}

	d_imp_consolidate_power =
	{
		icon = GFX_decision_imp_corruption
		days_remove = 90

		visible = 
		{
			is_ai = no
			OR = 
			{
    			check_variable = { imp_dip_sanctions_p > 10 }
				check_variable = { imp_mil_sanctions_p > 10 }
				check_variable = { imp_eco_sanctions_p > 10 }
				check_variable = { imp_fin_sanctions_p > 10 }
			}
		}

		available = {
			NOT = { has_idea = corruption_level_10 }
		}

		remove_effect = {
			increase_corruption = yes
			add_political_power = 300

			if = {
				limit = { has_government = fascism }
				add_popularity = { ideology = fascism popularity = 0.05}
			}

			if = {
				limit = { has_government = communism }
				add_popularity = { ideology = communism popularity = 0.05}
			}

			if = {
				limit = { has_government = nationalist }
				add_popularity = { ideology = nationalist popularity = 0.05}
			}

			if = {
				limit = { has_government = neutrality }
				add_popularity = { ideology = neutrality popularity = 0.05}
			}
		}
	}
}
